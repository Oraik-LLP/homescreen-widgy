cmake_minimum_required(VERSION 3.24)
project(panes VERSION 1.2.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(panes WIN32 src/app.cpp src/config.cpp src/features.cpp src/wallpaper.cpp src/render.cpp assets/app.rc)
target_include_directories(panes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_compile_definitions(panes PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(panes PRIVATE d2d1 dwrite dwmapi user32 gdi32 shell32 shlwapi ole32 comctl32 advapi32 wtsapi32 shcore psapi uuid windowscodecs comdlg32 msimg32)
if(MSVC)
  target_compile_options(panes PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(panes PRIVATE /MANIFEST:NO)
  set_property(TARGET panes PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(panes PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(panes PRIVATE -municode -static)
endif()
enable_testing()
add_executable(CoreTests tests/core_tests.cpp src/config.cpp src/features.cpp src/wallpaper.cpp)
target_compile_definitions(CoreTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(CoreTests PRIVATE shell32 ole32 uuid windowscodecs psapi)
if(MSVC)
  target_compile_options(CoreTests PRIVATE /W4 /permissive- /utf-8 /EHsc)
else()
  target_link_options(CoreTests PRIVATE -municode -static)
endif()
add_test(NAME core COMMAND CoreTests "${CMAKE_CURRENT_BINARY_DIR}/test-output")
